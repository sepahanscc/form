// =============================== // SEND TO GOOGLE SHEET (اصلاح شده) // =============================== async function sendToGoogleSheet(data) { try { // داده رو به فرمت مورد نظر Google Apps Script تبدیل می‌کنیم const payload = { action: "save", id: data.id || "", name: data.name || "", age: data.age || "", position: data.position || "", date: data.date || "", video: data.video || "", evaluator: data.evaluator || "", intel: data.scores?.intel || 0, tech: data.scores?.tech || 0, tact: data.scores?.tact || 0, phys: data.scores?.phys || 0, stat: data.scores?.stat || 0, total: data.total || 0 }; console.log("📤 ارسال داده:", payload); const response = await fetch(GOOGLE_SCRIPT_URL, { method: "POST", headers: { "Content-Type": "application/json", }, body: JSON.stringify(payload) }); if (!response.ok) { throw new Error(`HTTP error! status: ${response.status}`); } const result = await response.json(); console.log("✅ نتیجه ذخیره:", result); if (result.status === "ok") { console.log("✅ داده با موفقیت ذخیره شد"); } else { console.error("❌ خطا در ذخیره:", result); } } catch (error) { console.error("❌ خطا در ارتباط با سرور:", error); alert("⚠️ خطا در ارتباط با سرور. داده به صورت محلی ذخیره شد."); } }
